/** * DVAS GitHub Deployment Automation Script (Node.js) * Publishes the DVAS Portal directly to GitHub Pages without terminal/bash commands. * * Pre-requisite: Install octokit -> npm install octokit * Replace 'YOUR_GITHUB_PERSONAL_ACCESS_TOKEN' with a token having 'repo' & 'pages' scope. */ import { Octokit } from "octokit"; import fs from "fs"; // 1. Initialize GitHub API Client const GITHUB_TOKEN = process.env.GITHUB_TOKEN || "YOUR_GITHUB_PERSONAL_ACCESS_TOKEN"; const octokit = new Octokit({ auth: GITHUB_TOKEN }); const REPO_NAME = "dvas-revenue-portal"; async function deployToGitHub() { try { // 2. Fetch authenticated user details const { data: user } = await octokit.rest.users.getAuthenticated(); const username = user.login; console.log(`Authenticated as GitHub user: ${username}`); // 3. Create public repository via API console.log(`Creating repository '${REPO_NAME}'...`); await octokit.rest.repos.createForAuthenticatedUser({ name: REPO_NAME, description: "Diverse Visual Arts Studio (DVAS) Revenue & Security Portal powered by IBM Bob", private: false, auto_init: true // Creates initial commit with main branch }); console.log(`Repository created: https://github.com/${username}/${REPO_NAME}`); // 4. Read the index.html content const fileContent = fs.readFileSync("./index.html", "utf8"); const contentEncoded = Buffer.from(fileContent).toString("base64"); // 5. Commit & Push index.html to main branch via API console.log("Uploading index.html to main branch..."); await octokit.rest.repos.createOrUpdateFileContents({ owner: username, repo: REPO_NAME, path: "index.html", message: "Initial commit: Deploy DVAS Portal via IBM Bob API", content: contentEncoded, branch: "main" }); console.log("File index.html successfully committed."); // 6. Enable GitHub Pages Deployment via API console.log("Enabling GitHub Pages..."); await octokit.rest.repos.createPagesSite({ owner: username, repo: REPO_NAME, source: { branch: "main", path: "/" // Deploy from root } }); const liveUrl = `https://${username}.github.io/${REPO_NAME}/`; console.log(`\n🎉 SUCCESS! Your live project link is: ${liveUrl}`); return liveUrl; } catch (error) { console.error("Deployment failed:", error.response ? error.response.data : error.message); } } deployToGitHub();
